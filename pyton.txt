result = []

def divider(a, b):
    if a < b:
        raise ValueError("a меньше b")
    if b > 100:
        raise IndexError("b больше 100")
    return a / b

data = {10: 2, 2: 5, "123": 4, 18: 0, (1, 2): 15, 8: 4}

for key in data:
    try:
        res = divider(key, data[key])
        result.append(res)
    except Exception as e:
        print(f"Перехвачено исключение: {type(e).__name__} ({e}) для ключа {key}")

print("\nИтоговый результат:", result)
